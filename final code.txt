BlynkTimer timer;

unsigned long lastAlertTime = 0;
unsigned long lastMotionTime = 0;
unsigned long detectHold = 0;

unsigned long buzzerStartTime = 0;
bool buzzerActive = false;
bool prevDetect = false;

long duration;
float distance;

// Camera
String camURL = "http://10.107.201.101/capture";


float getDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return 999;

  return duration * 0.034 / 2.0;
}

bool checkVibration() {
  int count = 0;
  int threshold = 500;

  for (int i = 0; i < 5; i++) {
    if (analogRead(VIB_PIN) > threshold) count++;
    delay(10);
  }
  return (count >= 2);
}
void triggerCamera() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  http.begin(camURL);
  http.GET();
  http.end();
}

void setup() {

  Serial.begin(115200);

  pinMode(PIR_PIN, INPUT);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(VIB_PIN, INPUT);

  pinMode(BUZZER_PIN, OUTPUT);   
  digitalWrite(BUZZER_PIN, LOW);

  digitalWrite(TRIG_PIN, LOW);

  Blynk.begin(BLYNK_AUTH_TOKEN, ssid, pass);

  timer.setInterval(5000L, sendData);
}

// ================= LOOP =================
void loop() {
  Blynk.run();
  timer.run();

  // 🔔 Stop buzzer after 3 sec (ADDED)
  if (buzzerActive && millis() - buzzerStartTime >= 3000) {
    digitalWrite(BUZZER_PIN, LOW);
    buzzerActive = false;
  }
}
